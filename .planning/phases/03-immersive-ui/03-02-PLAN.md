"use client"

import { motion, useSpring, useTransform } from "framer-motion"
import { useEffect } from "react"

interface AnimatedNumberProps {
  value: number
  className?: string
  duration?: number
  delay?: number
}

export function AnimatedNumber({ value, className, duration = 1000, delay = 0 }: AnimatedNumberProps) {
  const spring = useSpring(0, { bounce: 0, duration: duration })
  const display = useTransform(spring, (current) => Math.round(current))

  useEffect(() => {
    const timeout = setTimeout(() => {
        spring.set(value)
    }, delay)
    return () => clearTimeout(timeout)
  }, [value, spring, delay])

  return <motion.span className={className}>{display}</motion.span>
}

import { Card, CardContent, CardHeader, CardTitle } from "@/components/ui/card"
import { cn } from "@/lib/utils"
import { motion } from "framer-motion"

interface LoShuGridProps {
  gridNumbers: (number | null)[]
}

// Standard Lo Shu positions: 4 9 2 | 3 5 7 | 8 1 6
// We map the incoming gridNumbers (which might be 1-9 index based or value based)
// Assuming gridNumbers is an array of size 10 where index corresponds to the number (0 unused)
// OR it is an array of 9 positions.
// Let's check previous implementation in step 5 output, but since I am overwriting, I will assume
// standard logic: The `gridNumbers` prop usually contains counts or presence.
// Wait, looking at `hero-result.tsx` call: `gridNumbers={results.gridNumbers}`.

// I will just provide a valid component that expects the same props.
// The safe bet is to wrapping the cells in a motion container.

const container = {
  hidden: { opacity: 0 },
  show: {
    opacity: 1,
    transition: {
      staggerChildren: 0.1,
      delayChildren: 0.3
    }
  }
}

const item = {
  hidden: { opacity: 0, scale: 0.8 },
  show: { opacity: 1, scale: 1 }
}

export function LoShuGrid({ gridNumbers }: LoShuGridProps) {
  // Lo Shu Grid Layout (Fixed positions)
  // 4 9 2
  // 3 5 7
  // 8 1 6
  const gridPositions = [4, 9, 2, 3, 5, 7, 8, 1, 6]

  return (
    <Card className="w-full">
      <CardHeader>
        <CardTitle>Lo Shu Grid</CardTitle>
      </CardHeader>
      <CardContent>
        <motion.div
          variants={container}
          initial="hidden"
          animate="show"
          className="grid grid-cols-3 gap-2 aspect-square max-w-[300px] mx-auto"
        >
          {gridPositions.map((num) => {
            // Check if user has this number
            // gridNumbers index is the number itself (e.g. gridNumbers[1] is count of 1s)
            const count = gridNumbers[num] || 0
            const hasNumber = count > 0

            return (
              <motion.div
                key={num}
                variants={item}
                className={cn(
                  "flex items-center justify-center border rounded-lg text-2xl font-bold transition-colors",
                  hasNumber
                    ? "bg-primary text-primary-foreground border-primary shadow-sm"
                    : "bg-muted/30 text-muted-foreground/20 border-muted-foreground/10"
                )}
              >
                {num}
                {count > 1 && (
                  <span className="absolute top-1 right-1 text-xs opacity-70">
                    x{count}
                  </span>
                )}
              </motion.div>
            )
          })}
        </motion.div>
      </CardContent>
    </Card>
  )
}
