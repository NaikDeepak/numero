3. Create Firebase initialization module (src/lib/firebase.ts):
   ```typescript
   import { initializeApp, getApps, type FirebaseApp } from "firebase/app"
   import { getAuth, type Auth } from "firebase/auth"
   import { getFirestore, type Firestore } from "firebase/firestore"

   // Firebase configuration from environment variables
   const firebaseConfig = {
     apiKey: process.env.NEXT_PUBLIC_FIREBASE_API_KEY,
     authDomain: process.env.NEXT_PUBLIC_FIREBASE_AUTH_DOMAIN,
     projectId: process.env.NEXT_PUBLIC_FIREBASE_PROJECT_ID,
     storageBucket: process.env.NEXT_PUBLIC_FIREBASE_STORAGE_BUCKET,
     messagingSenderId: process.env.NEXT_PUBLIC_FIREBASE_MESSAGING_SENDER_ID,
     appId: process.env.NEXT_PUBLIC_FIREBASE_APP_ID,
   }

   // Singleton pattern - only initialize if no apps exist
   // This prevents "Firebase app already exists" errors during hot module replacement
   let app: FirebaseApp
   let auth: Auth
   let db: Firestore

   if (getApps().length === 0) {
     app = initializeApp(firebaseConfig)
     auth = getAuth(app)
     db = getFirestore(app)
   } else {
     app = getApps()[0]
     auth = getAuth(app)
     db = getFirestore(app)
   }

   export { app, auth, db }
   ```

4. Create .env.example with placeholders:
   ```bash
   cat > .env.example << 'EOF'
   # Firebase Configuration (Client-side - safe to expose)
   NEXT_PUBLIC_FIREBASE_API_KEY=your_api_key_here
   NEXT_PUBLIC_FIREBASE_AUTH_DOMAIN=your_project_id.firebaseapp.com
   NEXT_PUBLIC_FIREBASE_PROJECT_ID=your_project_id
   NEXT_PUBLIC_FIREBASE_STORAGE_BUCKET=your_project_id.appspot.com
   NEXT_PUBLIC_FIREBASE_MESSAGING_SENDER_ID=your_sender_id
   NEXT_PUBLIC_FIREBASE_APP_ID=your_app_id

   # Server-side Only (NEVER use NEXT_PUBLIC_ prefix for secrets)
   # GEMINI_API_KEY=your_gemini_key_here (Phase 4)
   EOF
   ```

5. Verify .env.local is in .gitignore:
   ```bash
   cat .gitignore | grep ".env.local"
   ```
   If missing, add it:
   ```bash
   echo ".env.local" >> .gitignore
   ```

**Why singleton pattern:** Prevents Firebase re-initialization during Next.js hot module replacement in dev mode.

**Why NEXT_PUBLIC_ prefix:** Makes variables accessible in browser. Firebase client SDK config is NOT sensitive (API key is restricted in Firebase Console, not secret).

**IMPORTANT:** .env.local will be created after user provides Firebase credentials (user_setup step).
